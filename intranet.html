// POPULATE PUBLIC BALLOT FROM IEC CANDIDATE DATABASE function loadDynamicBallotOptions() { const candidates = JSON.parse(localStorage.getItem('melview_election_candidates')) || { governor: [ { name: "Arthur Vance", party: "Progressive Unity" }, { name: "Marcus Sterling", party: "Constitutional Law & Order" } ], senate: [ { name: "Rachel Sterling", party: "Judicial Reform" }, { name: "John Miller", party: "Civil Liberties & Commerce" }, { name: "Thomas Cole", party: "Public Safety Caucus" } ] }; const govSelect = document.getElementById('voteGov'); if (govSelect) { govSelect.innerHTML = ''; candidates.governor.forEach(c => { const opt = document.createElement('option'); opt.value = `${c.name} (${c.party})`; opt.textContent = `${c.name} (${c.party})`; govSelect.appendChild(opt); }); const writeIn = document.createElement('option'); writeIn.value = "Write-In / Abstain"; writeIn.textContent = "Write-In / Abstain"; govSelect.appendChild(writeIn); } const senSelect = document.getElementById('voteSenate'); if (senSelect) { senSelect.innerHTML = ''; candidates.senate.forEach(c => { const opt = document.createElement('option'); opt.value = `${c.name} (${c.party})`; opt.textContent = `${c.name} (${c.party})`; senSelect.appendChild(opt); }); } } document.addEventListener("DOMContentLoaded", function() { loadDynamicBallotOptions(); });
